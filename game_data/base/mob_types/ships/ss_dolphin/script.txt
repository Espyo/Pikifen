init {
    set_var green_light_tick false
}

script {
    idling {
        on_enter {
            play_sound pistons
            play_sound hum
            set_timer 0.01
        }
        on_timer {
            start_particles ss_dolphin_blue_light 40 -51 250
            start_particles ss_dolphin_blue_light 40 51 250
            if $green_light_tick = true
                start_particles ss_dolphin_green_light 0 0 250
                play_sound beep
                set_var green_light_tick false
            else
                set_var green_light_tick true
            end_if
            set_timer 0.75
        }
    }
}
