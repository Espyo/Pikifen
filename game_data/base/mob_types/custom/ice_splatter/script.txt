first_state = idling

script {
    idling {
        on_enter {
            set_timer 5
            set_animation appearing
            start_particles ice_splatter 0 0 16
            play_sound ice sound_id
        }
        on_timer {
            set_state disappearing
        }
    }
    disappearing {
        on_enter {
            set_animation disappearing
            stop_particles
            stop_sound $sound_id
        }
        on_animation_end {
            delete
        }
    }
}
