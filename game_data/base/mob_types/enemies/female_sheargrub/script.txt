first_state = hidden
death_state = dying

init {
	set_shadow_visibility false
}

script {
    hidden {
        on_enter {
            set_animation hidden
            set_hiding true
            set_near_reach chase
            set_far_reach chase
        }
        on_opponent_in_reach {
            set_state emerging
        }
        on_leave {
            set_hiding false
        }
    }
    
    emerging {
        on_enter {
			move_to_relative 20 0 0
			start_particles sheargrub_digging 0 0 10
            set_animation emerging
        }
		on_frame_signal {
			get_event_info sgnl frame_signal
			
			if $sgnl = 0
				set_radius 10
				set_shadow_visibility true
			else_if $sgnl = 1
				set_radius 12
			else_if $sgnl = 2
				set_radius 14
			end_if
		}
        on_animation_end {
			stop
			stop_particles
            set_state examining
        }
    }
    
    examining {
        on_enter {
            set_animation examining
        }
        on_opponent_in_reach {
            focus trigger
            set_state chasing
        }
        on_animation_end {
            set_state going_home
        }
        on_pikmin_land {
            set_state crushed
        }
        on_touch_hazard {
            get_event_info temp hazard
            if $temp = water
                receive_status drowning_enemy
            end_if
        }
    }
    
    burying {
        on_enter {
			move_to_relative 20 0 0
			start_particles sheargrub_digging 12 0 10
            set_animation burying
        }
		on_frame_signal {
			get_event_info sgnl frame_signal
			
			if $sgnl = 2
				set_radius 12
			else_if $sgnl = 1
				set_radius 8
			else_if $sgnl = 0
				stop_particles
				set_shadow_visibility false
			end_if
		}
        on_animation_end {
			set_radius 14
            set_state hidden
        }
    }
    
    chasing {
        on_enter {
            set_animation moving normal true
            move_to_target focused_mob
        }
        on_focus_off_reach {
            set_state going_home
        }
        on_far_from_home {
            set_state going_home
        }
        on_pikmin_land {
            set_state crushed
        }
        on_touch_hazard {
            get_event_info temp hazard
            if $temp = water
                receive_status drowning_enemy
            end_if
        }
    }
    
    going_home {
        on_enter {
            set_animation moving normal true
            move_to_target home
        }
        on_reach_destination {
            set_state burying
        }
        on_pikmin_land {
            set_state crushed
        }
        on_touch_hazard {
            get_event_info temp hazard
            if $temp = water
                receive_status drowning_enemy
            end_if
        }
    }
    
    crushed {
        on_enter {
            set_state dying
        }
    }
    
    dying {
        on_enter {
            start_dying
            set_animation dying
        }
        on_animation_end {
            finish_dying
        }
    }
    
}
